from __future__ import annotations

MAX_ATTEMPTS = 3
BASE_DELAY_SECONDS = 2


def can_retry(attempt: int) -> bool:
    return attempt < MAX_ATTEMPTS


def retry_delay(attempt: int) -> int:
    return BASE_DELAY_SECONDS * attempt
